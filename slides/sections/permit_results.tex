\begin{frame}[label=permit-spec-main]
    \frametitle{Event Study: Permit Activity After Ward Reassignment}
    \small
    \[
      \log\mathbb{E}[Y_{by}\mid\cdot]=\mu_b+\lambda_{g(b),y}
      +\sum_{t\neq-1}\beta_t Z_b\mathbf{1}\{y-2015=t\}.
    \]
    \begin{itemize}[itemsep=0.35em]
      \item $Y_{by}$: annual high-discretion permit count, 2010-2020. Estimated by PPML.
      \item $Z_b=+1$ for assignment to more stringent aldermen, $-1$ for more lenient, and $0$ for unchanged blocks.
      \item \textbf{Fixed effects}: block $\mu_b$ and ward pair $\times$ year $\lambda_{g(b),y}$.
    \end{itemize}
    Parallel trends: absent reassignment, nearby switched and unchanged blocks would have followed comparable permit trends.
    \vfill\hfill\hyperlink{permit-balance-history}{\beamergotobutton{Pre-period balance}}
\end{frame}
