% APPENDIX: NEW-CONSTRUCTION DATA
% ==============================================================================
\begin{frame}[label=construction-data-appendix]
    \frametitle{Building the New-Construction Data}
    \small
    \begin{enumerate}[itemsep=0.55em]
      \item \textbf{Permits.} Start from the City of Chicago permits issued 2006--2022 that erect a new building (26{,}921) and keep those for new residential buildings.
      \item \textbf{Assessor records.} Link each permit to the Cook County Assessor records of the new building on its parcels (houses, condominiums, apartment buildings), which report dwelling units, lot area and floor area.
      \item \textbf{One row per building.} Permits reaching the same building are combined; homes built under one permit are counted separately. Buildings completed in 2006--2007 under earlier permits enter from Assessor records.
      \item \textbf{Checks.} Ambiguous matches are resolved by hand; density is withheld when records disagree with the permit or are implausible.
    \end{enumerate}
    \medskip
    \textbf{Result:} 15{,}350 new residential buildings, 2006--2022, with dwelling units and lot area (5{,}322 multifamily), each dated, located and assigned to its ward and nearest boundary segment.

\end{frame}
