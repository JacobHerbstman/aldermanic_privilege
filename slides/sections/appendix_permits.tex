\begin{frame}[label=permit-separate-spec]
    \frametitle{Allowing the Two Reassignment Directions to Differ}
    \small
    \[
    \begin{aligned}
      \log\mathbb E[Y_{by}\mid\cdot]&=\mu_b+\lambda_{g(b),y}\\
       &+\sum_{t\neq-1}\left[\beta_t^{H}H_b+\beta_t^{L}L_b\right]\mathbf1\{y-2015=t\}.
    \end{aligned}
    \]
    \begin{itemize}[itemsep=0.5em]
      \item $H_b=1$ for reassignment toward greater stringency; $L_b=1$ for reassignment toward greater leniency. Both are zero for unchanged blocks, so $Z_b=H_b-L_b$.
      \item Same 2010--2020 sample, 2014 reference year, block and ward-pair $\times$ year fixed effects, and ward-pair clustering.
      \item Pooled estimates replace the event-time interactions with $H_b\mathbf1\{y\geq2015\}$ and $L_b\mathbf1\{y\geq2015\}$.
      \item The combined model restricts $\beta_t^H=-\beta_t^L$. The pooled equal-and-opposite test is $H_0:\beta^H+\beta^L=0$.
    \end{itemize}

\end{frame}
