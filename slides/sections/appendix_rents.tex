\begin{frame}[label=price-controls]
    \frametitle{Housing and Location Controls in the Price Comparisons}
    \small
    \textbf{Listed rents}
    \begin{itemize}[itemsep=0.3em]
      \item Log floor area, bedroom categories, and log bathrooms.
      \item Rental building-type and segment $\times$ month fixed effects.
      \item Clean-location sample; median rent for each floorplan-month.
    \end{itemize}
    \medskip
    \textbf{Home sale prices}
    \begin{itemize}[itemsep=0.3em]
      \item Log floor area, log land area, log building age, log bedrooms, log bathrooms, and garage indicator.
      \item Assessor property-class and segment $\times$ quarter fixed effects.
    \end{itemize}
    \medskip
    \textbf{Both}: distances to the nearest school, park, major road, and CTA stop, and to Lake Michigan. Outcomes are in 2022 dollars; standard errors clustered by ward pair.

\end{frame}
