\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.88\textwidth]{../tasks/price_boundary_results/output/price_boundary_placebos.pdf}
    \caption{Placebo Cutoffs for Listed Rents and Home Sale Prices}
    \figurenotes{The placebo windows do not overlap the main 500~ft sample, and neither rents nor sale prices change significantly across an artificial cutoff.
    The left column places an artificial cutoff 1000~ft inside the less-stringent ward; the right column places it 1000~ft inside the more-stringent ward.
    The top row reports listed rents and the bottom row reports home sale prices.
    Each model uses a 500~ft window around the artificial cutoff and otherwise follows Equation~\eqref{eq:price_rd}.
    Each point is the estimated log difference between a 100~ft distance band and the band immediately below the artificial cutoff.
    Shaded areas are 95\% confidence intervals.
    Standard errors are clustered by ward pair.}
    \label{fig:price_placebo_boundaries}
\end{figure}
